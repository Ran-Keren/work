const SPREADSHEET_ID = "1rKR1VaHaIDAUVgz_bWJ9nX42R_3uhl4KS9OZNojEYGU"; const TAB_NAME = "MAIN"; const DAY_COLUMNS = { "sunday": 3, // Column C "monday": 4, // Column D "tuesday": 5, // Column E "wednesday": 6, // Column F "thursday": 7, // Column G "friday": 8, // Column H "saturday": 9 // Column I }; function getTargetSheet() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(TAB_NAME); if (!sheet) { throw new Error("Tab not found: " + TAB_NAME); } return sheet; } function doGet(e) { try { const params = e ? e.parameter : {}; // Return the active tab name dynamically if (params.action === "meta") { const sheet = getTargetSheet(); return ContentService.createTextOutput(JSON.stringify({ status: "success", tabName: sheet.getName() })) .setMimeType(ContentService.MimeType.JSON); } const sheet = getTargetSheet(); const targetEmployee = params.employee; if (!targetEmployee) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No employee specified" })) .setMimeType(ContentService.MimeType.JSON); } const data = sheet.getDataRange().getValues(); let targetRow = -1; for (let i = 2; i < data.length; i++) { const sheetName = String(data[i][0] || "").trim(); if (sheetName.toLowerCase() === String(targetEmployee).toLowerCase()) { targetRow = i + 1; break; } } if (targetRow === -1) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Employee not found in column A starting from row 3" })) .setMimeType(ContentService.MimeType.JSON); } if (params.action === "save") { for (const [dayId, colIdx] of Object.entries(DAY_COLUMNS)) { const cell = sheet.getRange(targetRow, colIdx); const val = params[dayId]; if (val !== undefined && val !== "") { cell.setValue(Number(val)); } else { cell.setValue(""); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", action: "saved" })) .setMimeType(ContentService.MimeType.JSON); } let schedule = {}; for (const [dayId, colIdx] of Object.entries(DAY_COLUMNS)) { const cellVal = Number(data[targetRow - 1][colIdx - 1]); if (cellVal === 1) schedule[dayId] = "office"; else if (cellVal === 2 || cellVal === 4) schedule[dayId] = "home"; else if (cellVal === 3) schedule[dayId] = "night"; } return ContentService.createTextOutput(JSON.stringify({ status: "success", schedule: schedule })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { return doGet(e); }
